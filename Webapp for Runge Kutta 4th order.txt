SAFE_FUNCTIONS = {
    "sin": math.sin,
    "cos": math.cos,
    "tan": math.tan,
    "asin": math.asin,
    "acos": math.acos,
    "atan": math.atan,
    "sinh": math.sinh,
    "cosh": math.cosh,
    "tanh": math.tanh,
    "exp": math.exp,
    "log": math.log,     # natural log
    "log10": math.log10,
    "sqrt": math.sqrt,
    "abs": abs,
    "pi": math.pi,
    "e": math.e,
}


def build_safe_namespace(x_value, y_value):
    """Create a restricted namespace containing only safe math functions
    plus the current x and y values. No Python builtins are exposed."""
    namespace = dict(SAFE_FUNCTIONS)
    namespace["x"] = x_value
    namespace["y"] = y_value
    namespace["__builtins__"] = {}  # completely disable built-in functions
    return namespace


def evaluate_derivative(equation_text, x_value, y_value):
    """
    Safely evaluate the user-entered expression f(x, y) = dy/dx
    at a specific (x_value, y_value) point.

    Raises a ValueError with a clear, user-friendly message if the
    expression is invalid or produces an invalid mathematical result.
    """
    namespace = build_safe_namespace(x_value, y_value)
    try:
        result = eval(equation_text, namespace)
    except ZeroDivisionError:
        raise ValueError("The equation caused a division by zero during calculation.")
    except (NameError, SyntaxError):
        raise ValueError(
            "The equation contains an invalid or unsupported expression. "
            "Use only x, y, numbers, +, -, *, /, **, and functions like "
            "sin, cos, tan, exp, log, sqrt, abs."
        )
    except Exception as error:  # catch-all so the app never crashes
        raise ValueError(f"Could not evaluate the equation: {error}")

    if isinstance(result, complex):
        raise ValueError("The equation produced a complex number, which is not supported.")

    try:
        result = float(result)
    except (TypeError, ValueError):
        raise ValueError("The equation did not produce a valid numeric result.")

    if math.isnan(result) or math.isinf(result):
        raise ValueError("The equation produced an undefined result (NaN or infinity) "
                          "at one of the calculation points.")

    return result



def validate_inputs(equation_text, x0, y0, x_target, step_size):
    """
    Validate all user inputs and raise a ValueError with a clear message
    for any problem. Returns nothing if everything is valid.
    """
    if equation_text is None or equation_text.strip() == "":
        raise ValueError("Please enter a differential equation, for example: x + y")

    if step_size == 0:
        raise ValueError("Step size (h) cannot be zero.")

    direction = x_target - x0
    if direction == 0:
        raise ValueError("Target x must be different from the initial x0.")

    # The step size must point in the same direction as (target - x0)
    if (direction > 0 and step_size < 0) or (direction < 0 and step_size > 0):
        raise ValueError(
            "Step size direction is invalid: h must have the same sign as "
            "(target x - initial x0)."
        )

    number_of_steps = abs(direction) / abs(step_size)
    if number_of_steps > 100000:
        raise ValueError("Too many steps would be required. Please use a larger step size.")


def solve_with_rk4(equation_text, x0, y0, x_target, step_size):
    """
    Solve dy/dx = f(x, y) from x0 to x_target using the classic RK4 method.

    Returns:
        table_rows: list of dictionaries, one per step, containing
                    Step, x, y, k1, k2, k3, k4, New y
        x_values:   list of all x points visited (including the start)
        y_values:   list of all y points visited (including the start)
        final_x:    the actual last x value reached
        final_y:    the approximate y value at final_x
    """
    validate_inputs(equation_text, x0, y0, x_target, step_size)

    h = step_size
    x_current = x0
    y_current = y0

    # Number of full steps needed to (approximately) reach the target.
    raw_steps = (x_target - x0) / h
    number_of_steps = round(raw_steps)
    if number_of_steps < 1:
        number_of_steps = 1

    table_rows = []
    x_values = [x_current]
    y_values = [y_current]

    for step_index in range(1, number_of_steps + 1):
        # --- The four RK4 slope estimates ---
        k1 = h * evaluate_derivative(equation_text, x_current, y_current)
        k2 = h * evaluate_derivative(equation_text, x_current + h / 2, y_current + k1 / 2)
        k3 = h * evaluate_derivative(equation_text, x_current + h / 2, y_current + k2 / 2)
        k4 = h * evaluate_derivative(equation_text, x_current + h, y_current + k3)

        # --- Weighted average update ---
        y_next = y_current + (k1 + 2 * k2 + 2 * k3 + k4) / 6
        x_next = x_current + h

        table_rows.append({
            "Step": step_index,
            "x": round(x_current, 6),
            "y": round(y_current, 6),
            "k1": round(k1, 6),
            "k2": round(k2, 6),
            "k3": round(k3, 6),
            "k4": round(k4, 6),
            "New y": round(y_next, 6),
        })

        x_current, y_current = x_next, y_next
        x_values.append(x_current)
        y_values.append(y_current)

    # Warn (softly, via rounding note) if the target wasn't hit exactly
    final_x = x_current
    final_y = y_current
    return table_rows, x_values, y_values, final_x, final_y, raw_steps, number_of_steps



def create_solution_plot(x_values, y_values):
    """Build a clean, professional Matplotlib figure of the RK4 solution."""
    figure, axis = plt.subplots(figsize=(7, 4.5), dpi=120)

    axis.plot(x_values, y_values, color="#2563eb", linewidth=2.2,
              marker="o", markersize=4, markerfacecolor="#1d4ed8",
              label="RK4 numerical solution")

    axis.set_title("Runge-Kutta 4th Order Numerical Solution", fontsize=13, fontweight="bold")
    axis.set_xlabel("x", fontsize=11)
    axis.set_ylabel("y", fontsize=11)
    axis.grid(True, linestyle="--", alpha=0.5)
    axis.legend(loc="best", fontsize=9)
    figure.tight_layout()

    return figure


def run_rk4_solver(equation_text, x0_text, y0_text, x_target_text, h_text):
    """
    Main callback triggered by the 'Calculate RK4 Solution' button.
    Handles parsing, validation, calculation, and formatting of all outputs.
    Never raises an exception back to Gradio - all errors are caught and
    shown to the user as a friendly message instead.
    """
    empty_plot, _ = plt.subplots(figsize=(7, 4.5), dpi=120)
    empty_plot.text(0.5, 0.5, "No graph to display", ha="center", va="center")

    # ---- Parse numeric fields safely ----
    try:
        x0 = float(x0_text)
    except (TypeError, ValueError):
        return "**Error:** Initial x0 must be a valid number.", None, "", "", empty_plot

    try:
        y0 = float(y0_text)
    except (TypeError, ValueError):
        return "**Error:** Initial y0 must be a valid number.", None, "", "", empty_plot

    try:
        x_target = float(x_target_text)
    except (TypeError, ValueError):
        return "**Error:** Target x must be a valid number.", None, "", "", empty_plot

    try:
        h = float(h_text)
    except (TypeError, ValueError):
        return "**Error:** Step size h must be a valid number.", None, "", "", empty_plot

    equation_text = (equation_text or "").strip()

    # ---- Run the RK4 solver (validation happens inside) ----
    try:
        (table_rows, x_values, y_values,
         final_x, final_y, raw_steps, number_of_steps) = solve_with_rk4(
            equation_text, x0, y0, x_target, h
        )
    except ValueError as error:
        return f"**Error:** {error}", None, "", "", empty_plot

    plt.close(empty_plot)

    # ---- Build the final answer text ----
    note = ""
    if abs(raw_steps - number_of_steps) > 1e-6:
        note = (f"\n\n*Note: with h = {h}, the exact target x = {x_target} could not be "
                f"reached with whole steps. The closest value was used instead "
                f"({number_of_steps} steps), landing at x = {round(final_x, 6)}.*")

    final_answer_markdown = (
        f"### Final Answer\n\n"
        f"**y({round(final_x, 6)}) ≈ {round(final_y, 6)}**"
        f"{note}"
    )

    # ---- Build the step-by-step table ----
    results_dataframe = pd.DataFrame(table_rows)

    # ---- Build the explanation ----
    explanation_markdown = (
        "### How RK4 Calculated This Result\n\n"
        f"Starting from **x₀ = {x0}, y₀ = {y0}**, the method took **{number_of_steps} "
        f"step(s)** of size **h = {h}** to move toward x = {x_target}.\n\n"
        "For every step:\n"
        "- **k1** estimates the slope at the beginning of the interval.\n"
        "- **k2** estimates the slope at the midpoint, using k1 to predict y there.\n"
        "- **k3** re-estimates the midpoint slope, using k2 for a better prediction.\n"
        "- **k4** estimates the slope at the end of the interval, using k3.\n\n"
        "These four slope estimates are then combined into a single weighted-average "
        "slope: (k1 + 2·k2 + 2·k3 + k4) / 6, which is used to advance y to the next "
        "point. This weighting makes RK4 far more accurate than simpler methods "
        "like Euler's method, while still only requiring the function itself "
        "(no derivatives) to be evaluated."
    )

    # ---- Build the graph ----
    solution_figure = create_solution_plot(x_values, y_values)

    return final_answer_markdown, results_dataframe, explanation_markdown, "", solution_figure
